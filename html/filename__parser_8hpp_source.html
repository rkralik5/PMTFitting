59ParsedFile parseFilename(const std::string& fullFilename) {
60 ParsedFile result;
61
62 // Extract the filename without the path
63 result.path = fullFilename.substr(0, fullFilename.find_last_of('/'));
64 result.filename = fullFilename.substr(fullFilename.find_last_of("/")+1);
65 std::string baseName = result.filename.substr(0, result.filename.find_last_of('.'));
66
67 // Split the base name into tokens using underscore as a delimiter
68 std::vector<std::string> tokens = splitString(baseName);
69
70 // Loop over channels and extract information
71 // The regex checks if the first token is a channel (e.g. "Ch1")
72 // If it is, we expect the next three tokens to be manufacturer, model, and
73 // voltage, otherwise they are the first three tokens
74 size_t i = 0;
75 while (i + 3 < tokens.size()) {
76 FileInfo info;
77
78 if (tokens[i].substr(0, 2) == "Ch") {
79 info.channel = std::stof(tokens[i].substr(2));
80 info.manufacturer = tokens[i+1];
81 info.model = tokens[i+2];
82 if (std::tolower(tokens[i+3].back()) != 'v') {
83 throw std::invalid_argument("Invalid voltage format: "+tokens[i+3]);
84 }
85 info.voltage = std::stof(tokens[i+3].substr(0, tokens[i+3].size()-1));
86 i += 4;
87 } else {
88 info.manufacturer = tokens[i];
89 info.model = tokens[i+1];
90 if (std::tolower(tokens[i+2].back()) != 'v') {
91 throw std::invalid_argument("Invalid voltage format: "+tokens[i+2]);
92 }
93 info.voltage = std::stof(tokens[i+2].substr(0, tokens[i+2].size()-1));
94 i += 3;
95 }
96
97 result.devices.push_back(info);
98 }
99
100 // If there are any remaining tokens, they are part of the description
101 if (i < tokens.size()) {
102 std::string desc = tokens[i];
103 for (size_t j = i + 1; j < tokens.size(); ++j) {
104 desc += "_" + tokens[j];
105 }
106 result.description = desc;
107 } else {
108 result.description = "";
109 }
